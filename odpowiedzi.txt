-- 1. Tworzenie tabel w poprawnej kolejności
CREATE TABLE nauczyciel (id INT AUTO_INCREMENT PRIMARY KEY, imie VARCHAR(20), nazwisko VARCHAR(30));
CREATE TABLE uczen (id INT AUTO_INCREMENT PRIMARY KEY, imie VARCHAR(20), nazwisko VARCHAR(30), klasa VARCHAR(4), funkcja VARCHAR(15), data_urodzin DATE, adres VARCHAR(40));
CREATE TABLE przedmiot (id INT AUTO_INCREMENT PRIMARY KEY, nazwa VARCHAR(100), skrot VARCHAR(5), sala VARCHAR(5), liczba_godzin INT, nauczyciel_id INT, FOREIGN KEY (nauczyciel_id) REFERENCES nauczyciel(id));
CREATE TABLE ocena (id INT AUTO_INCREMENT PRIMARY KEY, ocena INT, uczen_id INT, przedmiot_id INT, FOREIGN KEY (uczen_id) REFERENCES uczen(id), FOREIGN KEY (przedmiot_id) REFERENCES przedmiot(id));
